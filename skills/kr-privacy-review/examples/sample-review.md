# 예시: 회원 엑셀 내보내기 PR 검토

에이전트가 보고서 형식과 근거 표기 수준을 맞추도록 돕는 예시입니다. 실제 검토에서는 대상 코드에 맞게 새로 작성합니다.

## 입력: PR diff (요약)

```python
# app/admin/routes.py
@router.get("/admin/members/export")
def export_members(user=Depends(current_user), db=Depends(get_db)):
    if not user.is_staff:                      # (1) 스태프면 누구나
        raise HTTPException(403)
    rows = db.query(Member).all()              # (2) 전체 컬럼, 전체 회원
    logger.info("export members by %s: %s", user.email, rows)   # (3) 로그에 데이터
    return csv_response(rows)                  # (4) ORM 객체 직렬화

# app/models.py
class Member(Base):
    email = Column(String)
    phone = Column(String)
    rrn = Column(String)                       # (5) 주민등록번호 평문
    password_hash = Column(String)
    marketing_opt_in = Column(Boolean)
```
