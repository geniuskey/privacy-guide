import { getCollection } from 'astro:content';
import SelfCheck from '../../components/SelfCheck.tsx';

export const docs = await getCollection('docs');
export const docTitles = Object.fromEntries(docs.map((d) => [d.id, d.data.title]));

새 기능이나 서비스를 설계할 때 아래 항목을 고르면, 확인해야 할 사항이 오른쪽(모바일은 아래)에 바로 정리됩니다.

- **입력값은 이 브라우저 안에서만 계산합니다.** 서버로 보내거나 저장하지 않고, 페이지를 닫으면 사라집니다. 이 사이트에는 분석·광고 스크립트도 없습니다.
- 결과는 **확인할 사항 목록**이지 법률적 승인이나 적합성 판단이 아닙니다. `review-required`가 붙은 항목은 공식 원문을 아직 대조하지 못한 내용입니다.

<SelfCheck client:load base={import.meta.env.BASE_URL} docTitles={docTitles} />
